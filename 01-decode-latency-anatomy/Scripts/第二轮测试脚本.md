# ============ 统一的压测脚本 ============
```bash
cat > /workspace/scripts/sweep.sh <<'EOF'
#!/bin/bash
# 用法: ./sweep.sh <port> <model> <tag>
#   vLLM      : ./sweep.sh 8000 Qwen/Qwen2.5-7B-Instruct                vllm
#   llama.cpp : ./sweep.sh 8080 /models/Qwen2.5-7B-Instruct-Q4_K_M.gguf llamacpp
PORT=$1; MODEL=$2; TAG=$3
TOKENIZER=Qwen/Qwen2.5-7B-Instruct      # ⚠️ 两边显式用同一个分词器
OUT=/workspace/results
IN_LEN=1024; OUT_LEN=256

bench () {  # $1=concurrency $2=num_prompts $3=seed $4=filename
  echo "=========== $TAG concurrency=$1 seed=$3 ==========="
  vllm bench serve \
    --backend openai --base-url http://localhost:$PORT --endpoint /v1/completions \
    --model "$MODEL" --tokenizer "$TOKENIZER" \
    --dataset-name random --random-input-len $IN_LEN --random-output-len $OUT_LEN \
    --random-range-ratio 0 \
    --ignore-eos --seed $3 \
    --temperature 0 \
    --num-prompts $2 --max-concurrency $1 \
    --percentile-metrics ttft,tpot,itl,e2el --metric-percentiles 50,90,99 \
    --save-result --result-dir $OUT --result-filename $4 || true
  sleep 5    # 让服务端排空,否则下一档的 TTFT 混入上一档的尾巴
}

# --- warmup:结果丢弃,只为把 CUDA graph / 分配器热起来 ---
echo ">>> WARMUP"
bench 8 32 999 ${TAG}_warmup.json

# --- 正式扫描:num-prompts 随并发涨,每档大约 8~10 轮批次 ---
bench 1   40  101 ${TAG}_c001.json
bench 2   60  102 ${TAG}_c002.json
bench 4   80  104 ${TAG}_c004.json
bench 8   120 108 ${TAG}_c008.json
bench 16  200 116 ${TAG}_c016.json
bench 32  320 132 ${TAG}_c032.json
bench 64  512 164 ${TAG}_c064.json

echo ">>> DONE. results in $OUT"
EOF
chmod +x /workspace/scripts/sweep.sh
```

监控脚本 sample_metrics.sh


```bash
#!/bin/bash
# 用法: ./sample_metrics.sh <输出csv> [采样间隔秒]
OUT=${1:-/workspace/results/metrics.csv}
INT=${2:-1}

cat > "$OUT" <<'HDR'
ts,running,waiting,kv_usage,preemptions,queue_sum,queue_cnt,prefill_sum,prefill_cnt,decode_sum,decode_cnt,iter_tok_sum,iter_tok_cnt,flops,read_bytes,gpu_mem_used_mib,gpu_util,power_w,temp_c
HDR

trap 'echo "sampler stopped"; exit 0' INT TERM

while true; do
    T0=$(date +%s.%N)
    M=$(curl -s --max-time 2 http://localhost:8000/metrics)
    g () { echo "$M" | grep -E "^vllm:$1(\{|\s)" | awk '{print $NF}' | head -1; }

    RUN=$(g num_requests_running)
    WAIT=$(g num_requests_waiting)
    KV=$(g kv_cache_usage_perc);      [ -z "$KV" ]  && KV=$(g gpu_cache_usage_perc)
    PRE=$(g num_preemptions_total);   [ -z "$PRE" ] && PRE=$(g num_preemptions)

    # histogram:采 _sum / _count,分析时做差分
    QS=$(g request_queue_time_seconds_sum);    QC=$(g request_queue_time_seconds_count)
    PS=$(g request_prefill_time_seconds_sum);  PC=$(g request_prefill_time_seconds_count)
    DS=$(g request_decode_time_seconds_sum);   DC=$(g request_decode_time_seconds_count)
    IS=$(g iteration_tokens_total_sum);        IC=$(g iteration_tokens_total_count)

    # 用来算实测算术强度,验证 12.3
    FL=$(g estimated_flops_per_gpu_total)
    RB=$(g estimated_read_bytes_per_gpu_total)

    SMI=$(nvidia-smi --query-gpu=memory.used,utilization.gpu,power.draw,temperature.gpu \
          --format=csv,noheader,nounits | tr -d ' ')

    echo "$T0,${RUN:-0},${WAIT:-0},${KV:-0},${PRE:-0},${QS:-0},${QC:-0},${PS:-0},${PC:-0},${DS:-0},${DC:-0},${IS:-0},${IC:-0},${FL:-0},${RB:-0},$SMI" >> "$OUT"

    # 补偿采样漂移:减去本轮实际耗时(用 awk,不依赖 bc)
    T1=$(date +%s.%N)
    SL=$(awk -v i="$INT" -v a="$T0" -v b="$T1" 'BEGIN{d=i-(b-a); print (d>0)?d:0}')
    sleep "${SL:-$INT}"
done
```

### 5.3 ⚠️ 一条消不掉的不对等,必须写进方法论

- **vLLM 的 `--max-model-len 8192` 是"每请求上限"**,KV 由 PagedAttention **按块按需**分配 ——
  86,896 token 的池子能同时装下 64 条声明 8192 上限的请求(实际每条只占 1280)。
- **llama.cpp 的 `-c ÷ -np` 是"每 slot 预留",必须预付** ——
  想让 64 条各有 8192,就得真的乘出来:64 × 8192 × 56 KiB = **28 GiB**,24 GB 卡装不下。

本轮负载每条只要 1280 token,两边都不受这个差异影响。
**但博文里不能把 `-c 262144` 和 `--max-model-len 8192` 当成同一个量来比。**

顺带一个副产品结论:llama.cpp 的 slot 是**硬分配**的,一条请求用不完也不能借给别人;
vLLM 的池子是**共享**的。同样的 token 总数,vLLM 利用率天然更高 ——
这是 PagedAttention 的价值,不是某一版实现的优劣。
